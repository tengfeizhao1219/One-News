### 4.4 Canvas 绘制参考

```javascript
function generateShareImage(category, isDark) {
  const colorMap = {
    tech:     isDark ? '#60A5FA' : '#007AFF',
    world:    isDark ? '#818CF8' : '#5856D6',
    sports:   isDark ? '#FBBF24' : '#FF9500',
    life:     isDark ? '#34D399' : '#34C759',
    recommend:isDark ? '#FF453A' : '#FF3B30',
    agriculture: isDark ? '#84CC16' : '#65A30D',
    science:  isDark ? '#A78BFA' : '#7C3AED',
  };
  const bg = colorMap[category] || colorMap.recommend;

  const canvas = wx.createOffscreenCanvas({ type: '2d', width: 400, height: 300 });
  const ctx = canvas.getContext('2d');

  // 背景
  ctx.fillStyle = bg;
  ctx.fillRect(0, 0, 400, 300);

  // 品牌文字
  ctx.fillStyle = '#FFFFFF';
  ctx.font = 'bold 36px "PingFang SC", sans-serif';
  ctx.textAlign = 'center';
  ctx.fillText('一页', 200, 140);

  // 分类名
  ctx.globalAlpha = 0.7;
  ctx.font = '20px "PingFang SC", sans-serif';
  ctx.fillText(`· ${getCategoryName(category)} ·`, 200, 180);

  return canvas.toDataURL('image/png');
}
```
